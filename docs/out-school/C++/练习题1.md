# 练习题1

``` c++
#include<bits/stdc++.h>
using namespace std;
bool p(int x)
{
    for(int i=2;i<x;i++) if(!(x%i)) return false;
    return true;
}
bool q(int x)
{
    return p(x+2)?true:false;
}
int main()
{
    int cnt=0;
    for(int i=1;i<=10;i++)
    {
        if(p(i)) continue;
        for(int j=i;j<=20;j++)
        {
            for(int k=1;k<=40;k++)
            {
                if(q(k)) continue;
                cnt++;
            }
            if(!(j%10)) break;
        }
        for(int jj=1;jj<=30;jj++)
        {
            if(jj>20) break;
            cnt++;
        }
    }
    cout<<"cnt="<<cnt<<endl;
    return 0;
}
```
