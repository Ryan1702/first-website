# 函数与数学函数

``` c++
#include<bits/stdc++.h>
using namespace std;
/*
函数的组成

返回类型 函数名(传入的参数)
{
    ...;
    return 返回值;(如果没有返回值就不用这句话，返回类型写void)
}
*/
int add(int x,int y)
{
    int sum=x+y;
    return sum;
}
int sub(int x,int y)
{
    return x-y;
}
int main()
{
    int a,b;
    cin>>a>>b;
    cout<<"两个数的和为"<<add(a,b)<<endl;
    cout<<"两个数的差为"<<sub(a,b)<<endl;

    // 数学函数部分 需要cmath头文件
    printf("%d\n",(int)floor(-4.3));// 向下取整
    printf("%d\n",(int)ceil(-4.3));// 向上取整
    printf("%d\n",(int)(-4.3));// 抹去小数部分
    printf("%d\n",(int)round(-4.3));// 四舍五入
    // 所有的数学函数都要在前面强制转换(int)然后再输出
    return 0;
}
```
