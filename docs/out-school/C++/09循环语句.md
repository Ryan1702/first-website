# 循环语句

``` c++
#include<bits/stdc++.h>
using namespace std;
int main()
{
    int a=1;
    while(a<=10) //一直运行
    {
        cout<<a<<endl;
        a++;
    }

    cout<<endl;

    a=1;
    do
    {
        cout<<a<<endl;
        a++;
    }while(a<=10);

    cout<<endl;

    for(int i=1;i<=10;i++)
    {
        cout<<i<<endl;
    }

    cout<<endl;

    // for(初始化;判断语句;赋值语句)

    // for(;;) 死循环

    int b=10;
    for(;b>=1;b--) cout<<b<<endl;

    cout<<endl;

    int cnt=0;
    for(int i=1;i<=10;i++)
    {
        for(int j=1;j<=20;j++)
        {
            for(int k=1;k<=40;k++)
            {
                cnt++;
            }
        }
        for(int jj=1;jj<=30;jj++)
        {
            cnt++;
        }
    }
    cout<<"cnt="<<cnt<<endl;

    cout<<endl;

    for(int i=1;i<=10;i++)
    {
        if(i==5) continue;
        cout<<i<<endl;
    }

    cout<<endl;

    for(int i=1;i<=10;i++)
    {
        if(i==5) break;
        cout<<i<<endl;
    }

    cout<<endl;

    // 补充一个条件语句
    char c='a';
    switch(c)
    {
        case 'a':
            printf("这是a\n");
            break;
        case 'b':
            printf("这是b\n");
            break;
        case 'c':
            printf("这是c\n");
            break;
        default:
            printf("未知\n");
    }

    return 0;
}
```
