# 作用域

``` c++
#include<bits/stdc++.h>
using namespace std;
int a,b,c; //全局变量，默认初始化为0，即这句话相当于int a=0,b=0,c=0;，作用域范围为第4行以后
int add(int x,int y)
{
    int d; // 局部变量，作用域范围为第7行至8行
    return x+y;
}
int sub(int x,int y)
{
    int e; // 局部变量，作用域范围从第12行至13行
    return x-y;
}
int main()
{
    int f=10; // 局部变量，作用域范围第17行至21行
    int d=1;
    cout<<a<<endl;
    cout<<d<<endl;
    return 0;
}
/*
作用域
一个变量的作用域范围是它当前所在的部分，不会跨越到其他部分
*/
```
