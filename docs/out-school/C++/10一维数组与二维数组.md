# 一维数组与二维数组

``` c++
#include<bits/stdc++.h>
using namespace std;
int a[3];// 定义一个数组，数组可以类比为数列
double b[5];// 后面方括号的数字n代表这个数组的大小（最多可以存储n个这种类型的数据）
// 数组中的所有数据必须是同一数据类型
int c[5][4];
int main()
{
    /*
    这n个数据分别可以通过下标索引的方式使用
    比如a[0],a[1],a[2],...,a[n-1]
    就是这n个数据组成了这些数组
    注意不一定这n个位置都要使用到，一般数组大小会多开大一点，防止数组越界
    */
    cout<<"请输入a数组（3个整数）：";
    for(int i=0;i<3;i++)
    {
        cin>>a[i];
    }
    cout<<"请输入b数组（5个小数）：";
    for(int i=0;i<5;i++) cin>>b[i];
    cout<<"a数组的第1位是"<<a[0]<<endl;
    cout<<"a数组的第2位是"<<a[1]<<endl;
    cout<<"b数组的奇数位的和是"<<b[0]+b[2]+b[4]<<endl;
    cout<<"b数组的偶数位的和是"<<b[1]+b[3]<<endl;

    int cnt=0;
    for(int i=0;i<5;i++)
    {
        for(int j=0;j<4;j++)
        {
            c[i][j]=++cnt;
        }
    }
    for(int i=0;i<5;i++)
    {
        for(int j=0;j<4;j++)
        {
            cout<<c[i][j]<<" ";
        }
        cout<<endl;
    }
    return 0;
}
```
