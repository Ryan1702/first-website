# 字符数组

``` c++
#include<bits/stdc++.h>
using namespace std;
char s[10]="ZYXZYX";// 数组大小最小需要开7，字符串长度+1
// 等同于char s[]={'Z','Y','X','Z','Y','X','\0'};
char t[]="WYFWYF";// 数组大小默认为7
int main()
{
    // int a;
    // a -> 10rmb
    // &a -> No.15

    // char s[100];
    // s[14] -> 10rmb
    // s -> No.1
    char a[100],b[200],c[300];
    cin>>a; //遇到空格或换行结束
    cout<<a<<endl;

    scanf("%s",b); //遇到空格或换行结束
    printf("%s\n",b);

    cin.getline(c,sizeof(c)); //输入整行 遇到换行结束

    cout<<s<<endl<<t<<endl;
    strcpy(s,t);
    cout<<"使用了strcpy之后："<<endl;
    cout<<s<<endl<<t<<endl;
    strcpy(s,"abc");
    cout<<"将前者变为abc"<<endl;
    cout<<s<<endl<<t<<endl;
    strcat(s,t);
    cout<<"使用了strcat之后"<<endl;
    cout<<s<<endl<<t<<endl;
    int lens=strlen(s),lent=strlen(t);
    cout<<"s的长度是"<<lens<<endl;
    cout<<"t的长度是"<<lent<<endl;
    if(strcmp(s,t)>0) cout<<"s大于t"<<endl;
    else if(strcmp(s,t)<0) cout<<"s小于t"<<endl;
    else cout<<"s与t完全一致"<<endl;
    return 0;
}
```
