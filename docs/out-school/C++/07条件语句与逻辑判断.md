# 条件语句与逻辑判断

``` c++
#include<bits/stdc++.h>
using namespace std;
int main()
{
    /*
    以下是一个条件语块可能出现的几种情况
    注意在做题的时候区分出每一个条件语句块便于理解

    1.单个条件判断，只在成立时生效
    if(条件)
    {
        ...;
        ...;
        ...;
    }

    2.单个条件判断，成立不成立都可执行相应部分
    if(条件)
    {
        ...;
    }
    else
    {
        ...;
        ...;
    }

    3.多个条件判断，按顺序只执行第一个能成立的部分
    if(a<=5)
    {
        ...;
    }
    else if(a<=10)
    {
        ...;
    }
    else
    {
        ...;
    }

    PS：可以在else前加若干个else if语句，并依旧符合第三种情况的标准
    */
    int a=10,b=10,c=5;
    if(a==b) printf("ab相等\n");
    if(a!=b) printf("ab不相等\n");
    if(a<=b) printf("a小于等于b\n");
    if(a<b) printf("a小于b\n");
    if(a>=b) printf("a大于等于b\n");
    if(a>b) printf("a大于b\n");

    bool i=true,j=false;
    if(i&&j) printf("i和j如果都为true结果才为true\n");
    if(i||j) printf("i和j如果有一个为true则结果为true\n");
    if(!i) printf("i取反（即true变false，false变true）后为true\n");
    //其实if条件所有语句中，括号中的内容，最后结果是返回一个bool类型的值
    //如果这个值是true则执行这部分语句，否则就跳过
    //所以以上的这些判断关系式比如==，!=，<=，>=，&&，||，!等等
    //这些最后都是返回一个bool类型的值
    return 0;
}
```
