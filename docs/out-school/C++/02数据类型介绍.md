# 数据类型介绍

``` c++
#include<bits/stdc++.h>
using namespace std;
int main()
{
    // 这里是单行注释

    /*
    这里是
    多行注释
    */
    
    cout<<"一个int类型的数据占用"<<sizeof(int)<<"字节"<<endl; //sizeof返回该类型占用的字节数
    cout<<"一个char类型的数据占用"<<sizeof(char)<<"字节"<<endl;
    cout<<"一个float类型的数据占用"<<sizeof(float)<<"字节"<<endl; // 精确到6-7位小数点
    cout<<"一个double类型的数据占用"<<sizeof(double)<<"字节"<<endl; // 精确到15-16位小数点
    cout<<"一个long long类型的数据占用"<<sizeof(long long)<<"字节"<<endl;
    int a;
    char b;
    float c;
    double d;
    long long e;
    // unsigned 无符号 存储空间大一倍
    // long 加长
    // short 缩短 只能修饰int

    /*
    1TB=1024GB（1024=2的10次方）
    1GB=1024MB
    1MB=1024KB
    1KB=1024Byte字节
    1Byte=8bit位

    进制学习
    最好要对二进制比较熟练
    */

    // C++中的等于号=是 -> 右边赋值给左边（一定要牢记这个概念）
    // 两个等于号== -> 才是数学中的等于

    a=10;
    b='a';
    c=2.3;
    d=3.4;
    e=2e10;
    // int f=2e10;
    cout<<a<<endl;
    // decimal 十进制 %d
    printf("这是int类型的数据：%d\n",a);
    printf("这是char类型的数据：%c\n",b);
    printf("这是float类型的数据：%f\n",c);
    printf("这是double类型的数据：%f\n",d);
    printf("这是long long类型的数据：%lld\n",e);
    return 0;
}
```
