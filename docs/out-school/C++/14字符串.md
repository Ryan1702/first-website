# 字符串

``` c++
#include<bits/stdc++.h>
using namespace std;
string s="ZYXZYX",t="WYFWYF";
int main()
{
    string a,b;
    cin>>a; //遇到空格或换行结束
    cout<<a<<endl;

    getline(cin,b); //输入整行 遇到换行结束

    cout<<s<<endl<<t<<endl;
    s=t;
    cout<<"t赋值给s"<<endl;
    cout<<s<<endl<<t<<endl;
    s="abc";
    cout<<"将前者变为abc"<<endl;
    cout<<s<<endl<<t<<endl;
    cout<<"s+t是"<<endl;
    cout<<s+t<<endl;
    cout<<"s的长度是"<<s.size()<<endl;
    cout<<"t的长度是"<<t.size()<<endl;
    if(s>t) cout<<"s大于t"<<endl;
    else if(s<t) cout<<"s小于t"<<endl;
    else cout<<"s与t完全一致"<<endl;
    return 0;
}
```
